# Everything is crawlable by default, so no blanket "Allow: /" line here: paired
# with a Disallow it makes the rule ambiguous for crawlers that take the first
# match rather than the longest one, and /thank-you then leaks into indexes.
User-agent: *
Disallow: /thank-you
